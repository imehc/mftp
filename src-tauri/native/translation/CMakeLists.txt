cmake_minimum_required(VERSION 3.22)
project(mftp_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
foreach(option METAL ACCELERATE BLAS OPENMP CPU_KLEIDIAI CUDA VULKAN RPC AVX512)
  set(GGML_${option} OFF CACHE BOOL "" FORCE)
endforeach()
# x86 SIMD defaults to the portable baseline; build.rs opts in per target.
foreach(option AVX AVX2 FMA F16C BMI2)
  set(GGML_${option} OFF CACHE BOOL "")
endforeach()
foreach(option COMMON TESTS TOOLS EXAMPLES SERVER APP)
  set(LLAMA_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${MFTP_LLAMA_SOURCE}" llama)
add_library(mftp-translation STATIC bridge.cpp)
target_link_libraries(mftp-translation PRIVATE llama)
install(TARGETS mftp-translation llama ggml ggml-cpu ggml-base ARCHIVE DESTINATION lib)
